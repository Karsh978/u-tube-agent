function downloadMarkdown(data) {
  const mdContent = `
# ${data.title}

## Description
${data.description}

## Tags
${data.tags.join(', ')}

## Script Outline
${data.scriptOutline.map(item => `- ${item}`).join('\n')}
  `;

  const blob = new Blob([mdContent], { type: 'text/markdown' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `youtube-plan-${Date.now()}.md`;
  a.click();
}
